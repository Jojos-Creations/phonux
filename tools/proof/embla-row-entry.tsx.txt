// The oval drag's write, routed here by the proof's build. The control eases the row toward the pointer as Embla's own drag does:
// the target moves and the body eases the strip after it.
const easeEmbla = (embla: any, px: number): void => {
  const e = embla.internalEngine();
  const baseDuration = e.scrollBody.useBaseDuration;
  const baseFriction = e.scrollBody.useBaseFriction;
  baseDuration();
  baseFriction();
  e.target.set(-px);
  e.animation.start();
};
(globalThis as any).__ovalMove = writeEmbla;

function View({ name, k }: { name: string; k: number }) {
  const hue = (k * 47) % 360;
  const [count, setCount] = React.useState(0);
  counts[name] = count;
  const anchor = React.useRef<HTMLDivElement>(null);
  maxers[name] = usePanelMaximize(anchor) as any;
  React.useEffect(() => {
    mounts[name] = (mounts[name] ?? 0) + 1;
    setters[name] = setCount;
    mountedNow.add(name);
    return () => {
      mountedNow.delete(name);
    };
  }, []);
  const drag = usePanelDragProps();
  const presence = usePanelPresence();
  const page = (grabProps?: any) => (
    <PanelSizeBox>
      <Panel title={name} sx={{ height: '100%' }} {...(grabProps === undefined ? {} : { grabProps: { ...grabProps, 'data-grab': name } })}>
        <PanelFrame>
          <div ref={anchor} tabIndex={-1} data-box={name} data-presence={presence.reason} style={{ height: '100%', background: `linear-gradient(135deg,hsl(${hue} 55% 40%),hsl(${(hue + 60) % 360} 55% 22%))`, color: '#fff', font: '24px sans-serif', padding: 12, boxSizing: 'border-box' }}>
            {name} ({SLOTS[k]} slot{SLOTS[k] > 1 ? 's' : ''}) <span data-count={name}>{count}</span>
          </div>
        </PanelFrame>
      </Panel>
    </PanelSizeBox>
  );
  return GRAB ? <DragToClose {...drag} closeOnRelease={false}>{(grab) => page(grab)}</DragToClose> : page();
}
function ExtraView() {
  return <div data-box="x" tabIndex={-1} style={{ height: '100%', background: '#a52', color: '#fff', font: '24px sans-serif', padding: 12 }}>x</div>;
}
function RootView() {
  return <div data-box={ROOT} style={{ height: '100%', background: '#243', color: '#fff', font: '24px sans-serif', padding: 12 }}>{ROOT}</div>;
}

// mount: false, as the app passes: this page renders the row itself. Without it the auto mount renders a second row that fails to attach.
const ws = createWorkspace({
  mount: false,
  root: definePanel({ id: ROOT, view: RootView as any, settings: { title: 'History', frame: 'none', ...(ROOT_MOVE ? { movable: true } : {}) } as any }),
  // `x`: a page opened later, any number of times (`__g.openExtra()`), one slot wide, drawn as the others are.
  templates: [...ids.map((id, k) => definePanel({ id, view: ((props: any) => <View {...props} name={id} k={k} />) as any, settings: { title: id, slots: SLOTS[k], maxSlots: 64, ...(PARK ? { overflow: 'park', capacityReturn: 'manual', restoreVisibility: 'live' } : {}) } as any })), definePanel({ id: 'x', view: ExtraView as any, settings: { title: 'x', multiple: true } as any })],
  tuning: { rowGesture: 'step', rowGap: GAP, ...(CAP === null ? {} : { maxLivePanels: CAP }) } as any,
});
for (const id of ids) ws.addPanel(id);

function App() {
  const ref = React.useRef<HTMLDivElement>(null);
  const rowRef = React.useRef<HTMLDivElement>(null);
  const [w, setW] = React.useState(0);
  React.useLayoutEffect(() => {
    const el = ref.current!;
    const ro = new ResizeObserver(() => setW(el.clientWidth));
    // `late=1`: the row is drawn unmeasured for a while first, as a host that measures after its first paint draws it.
    const late = q.get('late') === '1' ? setTimeout(() => ro.observe(el), 300) : (ro.observe(el), undefined);
    if (late === undefined) setW(el.clientWidth);
    return () => {
      clearTimeout(late);
      ro.disconnect();
    };
  }, []);
  const p = useProvidePanels({ workspace: ws, rootId: ws.rootId, phoneWidth: PHONE, containerWidth: w, overflowContainerRef: ref, rowEngine: ENGINE } as any);
  latest = p;
  // How often the list's order changed: a drop commits the order once.
  const order = p.entries.map((e: any) => e.id).join(' ');
  if (order !== lastOrder) { if (lastOrder !== null) orderChanges += 1; lastOrder = order; }
  const host = createFakeHost({ panels: p });
  return (
    <HostProvider host={host}>
      <div style={{ width: '100vw', height: '100vh', overflow: 'hidden' }}>
        <Workspace rowEngine={ENGINE} rowNearSlots={NEAR} overflowContainerRef={ref} rowRef={rowRef} containerWidth={w} rowResizeSettled phoneWidth={PHONE} fixedColumnCount={p.fixedViewCount} liveWidths={p.liveWidths} rowScroll={p.rowScroll} rowNumbers={p.rowNumbers} workspace={ws}>
          <PanelRow entries={p.entries} rootId={ws.rootId} closingIds={NONE} enteringViaAnchorIds={NONE} rowNumbers={p.rowNumbers} maxLiveWidth={p.maxLiveWidth} workspace={ws} />
        </Workspace>
        {BAR === null ? null : <Bar p={p} scroller={ref} />}
      </div>
    </HostProvider>
  );
}

// A dot bar drawn as the app draws one: the oval under the dots, a frame that clips the track and keeps the oval inside it.
function Bar({ p, scroller }: { p: any; scroller: React.RefObject<HTMLDivElement | null> }) {
  const dots = usePanelDots({ rowNumbers: p.rowNumbers, entries: p.entries, scroller, tuning: p.tuning });
  const frameW = BAR === 0 || BAR === null ? dots.width : Math.min(BAR, dots.width);
  const offset = React.useRef(0);
  offset.current = Math.min(Math.max(offset.current, dots.oval.left + dots.oval.width - frameW), dots.oval.left, Math.max(0, dots.width - frameW));
  offset.current = Math.max(0, offset.current);
  const { style, ...bar } = dots.barProps;
  return (
    <div data-bar-frame="" style={{ position: 'fixed', zIndex: 100000, bottom: 20, left: `calc(50% - ${frameW / 2}px)`, width: frameW, height: 20, overflow: 'hidden', background: '#222' }}>
      <div data-bar-track="" {...bar} style={{ ...style, position: 'absolute', top: 0, height: 20, left: -offset.current, width: dots.width }}>
        <div {...dots.ovalProps} data-dot-oval="" style={{ position: 'absolute', top: 0, height: 20, left: dots.oval.left, width: dots.oval.width, border: '1px solid #fff', borderRadius: 10, boxSizing: 'border-box' }} />
        {dots.dots.map((d) => <button key={d.id} type="button" {...dots.dotProps(d.id)} style={{ position: 'absolute', top: 6, height: 8, left: d.left, width: d.width, padding: 0, border: 0, borderRadius: 4, background: '#aaa' }} />)}
      </div>
    </div>
  );
}

const container = (): HTMLDivElement => document.querySelector<HTMLDivElement>('[data-row-scroller]')!;
// The window the row is seen through: Embla's root, or the old row's grid band (the container less its padding, centered strip).
const windowRect = (): { left: number; width: number } => {
  const e = document.querySelector<HTMLElement>('[data-row-embla-window]');
  if (e !== null) {
    const r = e.getBoundingClientRect();
    return { left: r.left, width: r.width };
  }
  const n = latest.rowNumbers;
  const c = container().getBoundingClientRect();
  const fit = Math.max(1, Math.floor((n.windowWidth + GAP) / (PHONE + GAP)));
  const width = Math.min(n.stripWidth, fit * (PHONE + GAP) - GAP);
  return { left: c.left + GAP + n.rowMarginLeft, width };
};
// What is drawn: the Embla box's transform as the browser computes it this frame. Embla's own `location` moves in fixed 60 Hz steps
// and the drawn translate interpolates between the last two, so `location` can stand still in a frame that is drawn moving.
const pos = (): number => {
  const win = document.querySelector<HTMLElement>('[data-row-embla-window]');
  const box = win?.firstElementChild as HTMLElement | null | undefined;
  if (box == null) return rowScrollerFor(container()).scrollLeft();
  return -new DOMMatrixReadOnly(getComputedStyle(box).transform).m41;
};
const logical = (): number => {
  const e = emblaOf(container());
  return e !== undefined ? emblaScrollOf(e) : rowScrollerFor(container()).scrollLeft();
};

const frame = (t: number): void => {
  if (recording) {
    const win = windowRect();
    const panels: Record<string, { x: number; w: number }> = {};
    const named: Record<string, { x: number; w: number }> = {};
    for (const el of document.querySelectorAll<HTMLElement>('[data-row-slot]')) {

      const name = el.querySelector<HTMLElement>('[data-box]')?.dataset.box;
      if (name !== undefined) named[name] = panels[el.dataset.rowId!]!;
    }
    const oval = document.querySelector<HTMLElement>('[data-dot-oval]');
    const track = document.querySelector<HTMLElement>('[data-bar-track]');
    const bar = oval === null || track === null ? {} : { ovalLeft: oval.getBoundingClientRect().left - track.getBoundingClientRect().left, ovalScreen: oval.getBoundingClientRect().left, ovalW: oval.getBoundingClientRect().width };
    // Each page as drawn this frame: its size box under the scale a resize eases (the root has none: its box).
    const drawn: Record<string, { x: number; w: number }> = {};
    for (const el of document.querySelectorAll<HTMLElement>('[data-row-slot]')) {
      const name = el.querySelector<HTMLElement>('[data-box]')?.dataset.box ?? el.dataset.rowId!;
      const sized = el.querySelector<HTMLElement>('[data-panel-size-box]') ?? el.querySelector<HTMLElement>('[data-row-unloaded]') ?? el.querySelector<HTMLElement>('[data-box]') ?? el;

    for (const el of document.querySelectorAll<HTMLElement>('[data-row-unloaded]')) {
      const r = el.getBoundingClientRect();
      if (Math.min(r.right, win.left + win.width) - Math.max(r.left, win.left) > 1) blankIn.push(el.closest<HTMLElement>('[data-row-slot]')!.dataset.rowId!);
    }
    const grabbing = document.querySelector<HTMLElement>('[data-dragging="true"] [data-box], [data-panel-dragging] [data-box]');

    wheels = 0;
    pd = 0;
    pm = 0;
    pu = 0;
  }
  requestAnimationFrame(frame);
};
