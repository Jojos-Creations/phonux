// Proof page for embla-row-proof.ts: the real Workspace and PanelRow from useProvidePanels, on the engine the query names.
// Query: slots=1,3,1,2 (each panel's slots) engine=embla|own
import * as React from 'react';
import { createRoot } from 'react-dom/client';
import { HostProvider, PanelRow, Workspace, createWorkspace, definePanel, useProvidePanels, Panel, PanelFrame, PanelSizeBox, feedRowWheel } from 'LIB/index.js';
import { createFakeHost } from 'LIB/fakeHost.js';
import { rowScrollerFor } from 'LIB/rowScroller.js';
import { emblaOf, emblaScrollOf } from 'LIB/emblaRow.js';

const q = new URLSearchParams(location.search);
const SLOTS = (q.get('slots') ?? '1,1,1,1,1,1').split(',').map(Number);
const ENGINE = q.get('engine') === 'own' ? 'own' : 'embla';
const GAP = 16;
const PHONE = 390;
const ROOT = 'root';
const NONE = new Set<string>();
const ids = SLOTS.map((_, k) => `p${k + 1}`);
let latest: any;

function View({ name, k }: { name: string; k: number }) {
  const hue = (k * 47) % 360;
  return (
    <PanelSizeBox>
      <Panel title={name} sx={{ height: '100%' }}>
        <PanelFrame>
          <div data-box={name} style={{ height: '100%', background: `linear-gradient(135deg,hsl(${hue} 55% 40%),hsl(${(hue + 60) % 360} 55% 22%))`, color: '#fff', font: '24px sans-serif', padding: 12, boxSizing: 'border-box' }}>
            {name} ({SLOTS[k]} slot{SLOTS[k] > 1 ? 's' : ''})
          </div>
        </PanelFrame>
      </Panel>
    </PanelSizeBox>
  );
}
function RootView() {
  return <div data-box={ROOT} style={{ height: '100%', background: '#243', color: '#fff', font: '24px sans-serif', padding: 12 }}>{ROOT}</div>;
}

const ws = createWorkspace({
  root: definePanel({ id: ROOT, view: RootView as any, settings: { title: 'History', frame: 'none' } as any }),
  templates: ids.map((id, k) => definePanel({ id, view: ((props: any) => <View {...props} name={id} k={k} />) as any, settings: { title: id, slots: SLOTS[k], maxSlots: 64 } as any })),
  tuning: { rowGesture: 'step', rowGap: GAP } as any,
});
for (const id of ids) ws.addPanel(id);

function App() {
  const ref = React.useRef<HTMLDivElement>(null);
  const rowRef = React.useRef<HTMLDivElement>(null);
  const [w, setW] = React.useState(0);
  React.useLayoutEffect(() => {
    const el = ref.current!;
    const ro = new ResizeObserver(() => setW(el.clientWidth));
    ro.observe(el);
    setW(el.clientWidth);
    return () => ro.disconnect();
  }, []);
  const p = useProvidePanels({ workspace: ws, rootId: ws.rootId, phoneWidth: PHONE, containerWidth: w, overflowContainerRef: ref } as any);
  latest = p;
  const host = createFakeHost({ panels: p });
  return (
    <HostProvider host={host}>
      <div style={{ width: '100vw', height: '100vh', overflow: 'hidden' }}>
        <Workspace rowEngine={ENGINE} overflowContainerRef={ref} rowRef={rowRef} containerWidth={w} rowResizeSettled phoneWidth={PHONE} fixedColumnCount={p.fixedViewCount} liveWidths={p.liveWidths} rowScroll={p.rowScroll} rowNumbers={p.rowNumbers} workspace={ws}>
          <PanelRow entries={p.entries} rootId={ws.rootId} closingIds={NONE} enteringViaAnchorIds={NONE} rowNumbers={p.rowNumbers} maxLiveWidth={p.maxLiveWidth} workspace={ws} />
        </Workspace>
      </div>
    </HostProvider>
  );
}

const container = (): HTMLDivElement => document.querySelector<HTMLDivElement>('[data-row-scroller]')!;
// The window the row is seen through: Embla's root, or the old row's grid band (the container less its padding, centered strip).
const windowRect = (): { left: number; width: number } => {
  const e = document.querySelector<HTMLElement>('[data-row-embla-window]');
  if (e !== null) {
    const r = e.getBoundingClientRect();
    return { left: r.left, width: r.width };
  }
  const n = latest.rowNumbers;
  const c = container().getBoundingClientRect();
  const fit = Math.max(1, Math.floor((n.windowWidth + GAP) / (PHONE + GAP)));
  const width = Math.min(n.stripWidth, fit * (PHONE + GAP) - GAP);
  return { left: c.left + GAP + n.rowMarginLeft, width };
};
const pos = (): number => {
  const e = emblaOf(container());
  return e !== undefined ? emblaScrollOf(e) : rowScrollerFor(container()).scrollLeft();
};
const frames: any[] = [];
let recording = false;
let pointerX = 0;
// Wheel events since the last frame: a frame no wheel reached cannot move a row that follows its input exactly.
let wheels = 0;
window.addEventListener('wheel', () => (wheels += 1), { capture: true, passive: true });
window.addEventListener('pointermove', (e) => (pointerX = e.clientX), true);
const frame = (t: number): void => {
  if (recording) {
    const win = windowRect();
    const panels: Record<string, { x: number; w: number }> = {};
    for (const el of document.querySelectorAll<HTMLElement>('[data-row-slot]')) {
      const r = el.getBoundingClientRect();
      panels[el.dataset.rowId!] = { x: +(r.left - win.left).toFixed(2), w: +r.width.toFixed(2) };
    }
    frames.push({ t, pos: pos(), winW: win.width, winLeft: win.left, pointerX, wheels, panels });
    wheels = 0;
  }
  requestAnimationFrame(frame);
};
requestAnimationFrame(frame);

(window as any).__g = {
  engine: ENGINE,
  PITCH: PHONE + GAP,
  PHONE,
  GAP,
  ready: () => latest !== undefined && Number.isFinite(latest.rowNumbers?.windowWidth) && document.querySelectorAll('[data-row-slot]').length === ids.length + 1 && (ENGINE === 'own' || emblaOf(container()) !== undefined),
  pos,
  win: windowRect,
  max: () => Math.max(0, latest.rowNumbers.stripWidth - windowRect().width),
  stripWidth: () => latest.rowNumbers.stripWidth,
  ids: () => latest.rowNumbers.ids as string[],
  widths: () => latest.rowNumbers.widths as number[],
  leftPanel: () => {
    const win = windowRect();
    let best: string | null = null;
    let bx = Number.POSITIVE_INFINITY;
    for (const el of document.querySelectorAll<HTMLElement>('[data-row-slot]')) {
      const r = el.getBoundingClientRect();
      if (r.right - win.left > 1 && r.left - win.left < bx) { bx = r.left - win.left; best = el.dataset.rowId!; }
    }
    return best;
  },
  jumpToSlot: (s: number) => emblaOf(container())?.scrollTo(s, true),
  // A wheel handed in from embedded content, as a webview's script does: deltas only, from an element inside a panel.
  feed: (init: any) => feedRowWheel(document.querySelector('[data-box="p2"]')!, init),
  // A stream of handed-in wheels, one per animation frame, as fix1-wheel.mjs sends them.
  feedStream: (D: number, ms: number) => new Promise((done) => {
    const from = document.querySelector('[data-box="p2"]')!;
    let input = 0;
    const t0 = performance.now();
    const tick = (now: number): void => {
      if (now - t0 >= ms) return done(input);
      feedRowWheel(from, { deltaX: D });
      input += Math.abs(D);
      requestAnimationFrame(tick);
    };
    requestAnimationFrame(tick);
  }),
  record: () => { frames.length = 0; recording = true; },
  stop: () => { recording = false; return frames.slice(); },
};
createRoot(document.getElementById('root')!).render(<App />);
