// A row whose panels hold at an edge, as the app builds one: History as the scrolling root, panels of the given slots, and
// `holds=<id>:<left|right|both>,...` setting each panel's edgeHold. Each panel is a box with its id as its text, so a test reads
// where every panel is drawn from the page. Query: slots=1,1,1 holds=p1:left gap=16 gesture=off|step (default off)
import * as React from 'react';
import { createRoot } from 'react-dom/client';
import { HostProvider, PanelRow, Workspace, createWorkspace, definePanel, useProvidePanels } from 'LIB/index.js';
import { createFakeHost } from 'LIB/fakeHost.js';
import { maxScrollOf } from 'LIB/rowNumbers.js';
import { rowScrollerFor } from 'LIB/rowScroller.js';

const q = new URLSearchParams(location.search);
const SLOTS = (q.get('slots') ?? '1,1,1').split(',').map(Number);
const HOLDS = new Map((q.get('holds') ?? '').split(',').filter(Boolean).map((pair) => pair.split(':') as [string, string]));
const GAP = Number(q.get('gap') ?? 16);
const PHONE = 390;
const ROOT = 'root';
const NONE = new Set<string>();
const ids = SLOTS.map((_, k) => `p${k + 1}`);
let latest: any;

// Each view counts its own mounts and keeps its DOM node, so a test can tell a view that stayed mounted from one that was re-created.
function View({ panel, name }: { panel: any; name: string }) {
  const ref = React.useRef<HTMLDivElement>(null);
  React.useEffect(() => {
    const w = window as any;
    w.__mounts = w.__mounts ?? {};
    w.__mounts[name] = (w.__mounts[name] ?? 0) + 1;
    w.__nodes = w.__nodes ?? {};
    w.__nodes[name] = ref.current;
  }, []);
  return (
    <div ref={ref} data-box={name} style={{ position: 'relative', width: '100%', height: '100%', background: name === ROOT ? '#243' : '#335', color: '#fff' }}>
      {name}
    </div>
  );
}

const ws = createWorkspace({
  root: definePanel({ id: ROOT, view: ((props: any) => <View {...props} name={ROOT} />) as any, settings: { title: 'History', frame: 'none' } }),
  templates: ids.map((id, k) => definePanel({ id, view: ((props: any) => <View {...props} name={id} />) as any, settings: { slots: SLOTS[k], maxSlots: 64, frame: 'none', ...(HOLDS.has(id) ? { edgeHold: HOLDS.get(id) } : {}) } as any })),
  tuning: { rowGesture: (q.get('gesture') ?? 'off') as 'step' | 'off', rowGap: GAP } as any,
});
for (const id of ids) ws.addPanel(id);

function App() {
  const ref = React.useRef<HTMLDivElement>(null);
  const rowRef = React.useRef<HTMLDivElement>(null);
  const [w, setW] = React.useState(0);
  React.useLayoutEffect(() => {
    const el = ref.current!;
    const ro = new ResizeObserver(() => setW(el.clientWidth));
    ro.observe(el);
    setW(el.clientWidth);
    return () => ro.disconnect();
  }, []);
  const p = useProvidePanels({ workspace: ws, rootId: ws.rootId, phoneWidth: PHONE, containerWidth: w, overflowContainerRef: ref } as any);
  latest = p;
  const host = createFakeHost({ panels: p });
  (window as any).__e = { container: () => ref.current, provided: () => latest, offset: () => rowScrollerFor(ref.current!).scrollLeft(), maxScroll: () => maxScrollOf(latest.rowNumbers), moveTo: (x: number) => rowScrollerFor(ref.current!).glideTo(x, { durationSec: 0, instant: true, ease: 'linear' }) };
  return (
    <HostProvider host={host}>
      <div style={{ width: '100vw', height: '100vh', overflow: 'hidden' }}>
        <Workspace overflowContainerRef={ref} rowRef={rowRef} containerWidth={w} rowResizeSettled phoneWidth={PHONE} fixedColumnCount={p.fixedViewCount} liveWidths={p.liveWidths} rowScroll={p.rowScroll} rowNumbers={p.rowNumbers} tuning={p.tuning}>
          <PanelRow entries={p.entries} rootId={ws.rootId} closingIds={NONE} enteringViaAnchorIds={NONE} rowNumbers={p.rowNumbers} maxLiveWidth={p.maxLiveWidth} workspace={ws} />
        </Workspace>
      </div>
    </HostProvider>
  );
}
createRoot(document.getElementById('root')!).render(<App />);
